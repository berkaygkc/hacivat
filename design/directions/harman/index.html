-- Top customers by revenue, last 30 days
SELECT c.full_name,executed· rows
c.city, c.segment,
count(o.id) AS orders,bigint
sum(oi.line_total) AS revenuenumeric
FROM customers c
JOIN orders o ON o.customer_id = c.id
JOIN order_items oi ON oi.order_id = o.id
WHERE o.ordered_at > now() - interval '30 days'
GROUP BY c.full_name, c.city, c.segment
ORDER BY revenue DESC
LIMIT 100;
